steps=[]
for name,exe,filename in [('final-php','C:/xampp/php/php.exe','staging-final-php.log'),('final-mobile','node','staging-final-mobile.log')]:
 cmd=[exe]+m.EXPECTED_COMMAND_ARGS[name]
 start_ns=time.time_ns();start=datetime.datetime.now(datetime.timezone.utc).isoformat();t=time.perf_counter()
